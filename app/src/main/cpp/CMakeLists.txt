cmake_minimum_required(VERSION 3.22.1)
project(dontree LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)

set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.deps")
set(WHISPER_URL "https://codeload.github.com/ggml-org/whisper.cpp/tar.gz/refs/tags/v1.7.6")
set(CHROMAPRINT_URL "https://codeload.github.com/acoustid/chromaprint/tar.gz/refs/tags/v1.5.1")
if(EXISTS "${DEPS}/whisper.tar.gz")
    set(WHISPER_URL "${DEPS}/whisper.tar.gz")
endif()
if(EXISTS "${DEPS}/chromaprint.tar.gz")
    set(CHROMAPRINT_URL "${DEPS}/chromaprint.tar.gz")
endif()
FetchContent_Declare(whisper URL "${WHISPER_URL}" URL_HASH SHA256=166140e9a6d8a36f787a2bd77f8f44dd64874f12dd8359ff7c1f4f9acb86202e)
FetchContent_MakeAvailable(whisper)

FetchContent_Declare(chromaprint URL "${CHROMAPRINT_URL}" URL_HASH SHA256=a1aad8fa3b8b18b78d3755b3767faff9abb67242e01b478ec9a64e190f335e1c)
FetchContent_GetProperties(chromaprint)
if(NOT chromaprint_POPULATED)
    FetchContent_Populate(chromaprint)
endif()
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(FFT_LIB kissfft CACHE STRING "" FORCE)
set(AUDIO_PROCESSOR_LIB builtin CACHE STRING "" FORCE)
set(KISSFFT_ROOT "${chromaprint_SOURCE_DIR}/src/3rdparty/kissfft")
set(KISSFFT_SOURCE_DIR "${KISSFFT_ROOT}" CACHE PATH "Bundled KissFFT sources" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${chromaprint_SOURCE_DIR}" "${chromaprint_BINARY_DIR}" EXCLUDE_FROM_ALL)
# Keep LGPL Chromaprint dynamically linked and use an Android-compatible SONAME.
set_target_properties(chromaprint PROPERTIES VERSION "" SOVERSION "")

add_library(dontree SHARED native_audio.cpp)
target_include_directories(dontree PRIVATE "${chromaprint_SOURCE_DIR}/src")
target_link_libraries(dontree PRIVATE whisper chromaprint android log)
target_link_options(dontree PRIVATE "-Wl,-z,max-page-size=16384")
